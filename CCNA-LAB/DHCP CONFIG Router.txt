# interface config router

# Main interface configuration (trunk connection to the switch)
# No ip address on the main interface
# Ensure the interface is up

int gig xxxx
 no ip address             
 no shutdown               

# Subinterface for VLAN 10
# Encapsulation for VLAN 10
# Gateway for VLAN 10
# Ensure the subinterface is up

int gig0/1.10
 encapsulation dot1Q 10    
 ip address 192.168.10.1 255.255.255.192  
 no shutdown               

int gig0/1.20
 encapsulation dot1Q 20 
 ip address 192.168.20.1 255.255.255.128  
 no shutdown 

int gig0/1.30
 encapsulation dot1Q 30  
 ip address 192.168.30.1 255.255.254.0
 no shutdown 
 
int gig0/1.99
encapsulation dot1Q 99 
ip address 192.168.0.1 255.255.255.240
no shut

int gig0/1.999
encapsulation dot1Q 999
ip address 192.168.0.17 255.255.255.252
no shut




# Subinterface for VLAN 20
interface GigabitEthernet0/0.20
 encapsulation dot1Q 20    # Encapsulation for VLAN 20
 ip address 192.168.20.1 255.255.255.128  # Gateway for VLAN 20
 no shutdown               # Ensure the subinterface is up

# Subinterface for VLAN 30
interface GigabitEthernet0/0.30
 encapsulation dot1Q 30    # Encapsulation for VLAN 30
 ip address 192.168.30.1 255.255.254.0    # Gateway for VLAN 30
 no shutdown               # Ensure the subinterface is up

# Subinterface for Management VLAN (VLAN 99)
interface GigabitEthernet0/0.99
 encapsulation dot1Q 99    # Encapsulation for VLAN 99
 ip address 192.168.0.1   255.255.255.240   # Gateway for Management VLAN
 no shutdown               # Ensure the subinterface is up

# Subinterface for Native VLAN (VLAN 999, if needed for specific management or protocol)
interface GigabitEthernet0/0.999
 encapsulation dot1Q 999    # Encapsulation for Native VLAN
 ip address 192.168.0.17   255.255.255.252   # Gateway for Native VLAN
 no shutdown                # Ensure the subinterface is up
